function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("steps"); var data = JSON.parse(e.postData.contents); sheet.appendRow([ "", // Column A blank data.date || new Date(), // Column B: Date data.steps, // Column C: Steps data.stars, // Column D: Star Earned data.reward, // Column E: Rewards redeemed data.balance // Column F: Star balance ]); return ContentService.createTextOutput("Success"); } function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("steps"); var data = sheet.getDataRange().getValues(); var result = []; for (var i = 1; i < data.length; i++) { // skip header row let rawDate = data[i][1]; let dateStr; if (rawDate instanceof Date) { // Normalize to yyyy-MM-dd dateStr = Utilities.formatDate(rawDate, Session.getScriptTimeZone(), "yyyy-MM-dd"); } else { dateStr = rawDate; } result.push({ date: dateStr, // always a clean string now steps: data[i][2], stars: data[i][3], reward: data[i][4], balance: data[i][5] }); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
